async function uploadArquivoEspecial(event, tipo) { if (perfilUsuario !== 'ADM' && perfilUsuario !== 'LOGISTICA') { alert("Apenas os perfis Logística e ADM podem fazer o upload de anexos."); return; } const file = event.target.files[0]; const statusElement = document.getElementById(`status-upload-${tipo}`); if (!file || !dataTransporteSelecionada) { alert("Selecione uma data válida e um arquivo."); return; } statusElement.innerText = "A enviar arquivo para o banco..."; // Nome do arquivo mantendo extensão original const filePath = `${dataTransporteSelecionada}/${tipo}_${Date.now()}_${file.name}`; const { data: uploadData, error: uploadError } = await supabaseClient .storage .from('anexos_notas') .upload(filePath, file, { upsert: true }); if (uploadError) { if (uploadError.message.includes('Bucket not found')) { statusElement.innerText = "Erro: Crie o bucket 'anexos_notas' no painel do Supabase Storage."; } else { statusElement.innerText = "Erro ao subir arquivo: " + uploadError.message; } return; } // Obter URL pública do arquivo salvo const { data: publicUrlData } = supabaseClient .storage .from('anexos_notas') .getPublicUrl(filePath); const campoUrl = tipo === 'xml' ? 'url_xml' : 'url_pdf'; const { error: updateError } = await supabaseClient .from('dispositivos') .update({ [campoUrl]: publicUrlData.publicUrl }) .eq('data_transporte', dataTransporteSelecionada); if (updateError) { statusElement.innerText = "Erro ao vincular URL aos registros: " + updateError.message; } else { statusElement.innerText = `Arquivo ${file.name} anexado com sucesso!`; } }
